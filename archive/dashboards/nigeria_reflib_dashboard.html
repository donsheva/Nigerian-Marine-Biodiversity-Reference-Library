Script 1 — Query NCBI for existing 12S sequences
Searches NCBI Nucleotide for West African species. Run this first — no fieldwork needed. Saves results to ncbi_gap_report.csv
# ncbi_query.py
# Run: python ncbi_query.py

from Bio import Entrez, SeqIO
import pandas as pd, time

Entrez.email = "your@email.com"   # required by NCBI

SPECIES = [
    "Ethmalosa fimbriata",
    "Sardinella maderensis",
    "Pseudotolithus senegalensis",
    "Pseudotolithus elongatus",
    "Cynoglossus senegalensis",
    "Galeoides decadactylus",
    "Polydactylus quadrifilis",
    "Lutjanus agennes",
    "Arius latiscutatus",
    "Brachydeuterus auritus",
    "Elops lacerta",
    "Rhizoprionodon acutus",
    "Rhinobatos rhinobatos",
    "Hypanus marianae",
    "Penaeus notialis",
    "Callinectes amnicola",
    "Trichechus senegalensis",
    "Sousa teuszii",
]

records = []
for sp in SPECIES:
    query = f'"{sp}"[Organism] AND 12S[Title]'
    handle = Entrez.esearch(db="nucleotide", term=query)
    result = Entrez.read(handle); handle.close()
    count = int(result["Count"])
    print(f"{sp:45} {count:4} sequences")
    records.append({
        "species": sp,
        "ncbi_12S_count": count,
        "status": "present" if count > 2 else "partial" if count > 0 else "absent"
    })
    time.sleep(0.4)   # polite delay

df = pd.DataFrame(records)
df.to_csv("ncbi_gap_report.csv", index=False)
print(f"\nSaved: ncbi_gap_report.csv ({len(df)} species)")
print(df.groupby("status").size())
Script 2 — Query BOLD Systems public API
Downloads available barcodes from Nigeria and Gulf of Guinea. Saves a FASTA file you can use immediately in BLAST searches.
# bold_query.py
# Run: python bold_query.py

import requests, time, json

BASE = "http://www.boldsystems.org/index.php/API_Public/combined"

GROUPS = [
    ("Actinopterygii",  "ray-finned fishes"),
    ("Elasmobranchii",  "sharks and rays"),
    ("Malacostraca",    "shrimps and crabs"),
    ("Cetacea",         "dolphins and whales"),
    ("Testudines",      "sea turtles"),
    ("Trichechus",      "manatees"),
]

GEOS = ["Nigeria", "Ghana", "Benin", "Cameroon", "Togo"]

all_records = {}

for taxon, label in GROUPS:
    for geo in GEOS:
        params = {"taxon": taxon, "geo": geo, "format": "json"}
        try:
            r = requests.get(BASE, params=params, timeout=30)
            if r.status_code == 200 and r.text.strip():
                data = r.json()
                recs = data.get("bold_records", {})
                all_records.update(recs)
                print(f"  {taxon} / {geo}: {len(recs)} records")
        except Exception as e:
            print(f"  Error {taxon}/{geo}: {e}")
        time.sleep(0.8)

# Save as FASTA
fasta_lines = []
for rec_id, rec in all_records.items():
    seq = rec.get("sequences", {}).get("sequence", "")
    sp  = rec.get("taxonomy", {}).get("species", {}).get("taxon", {}).get("name", "unknown")
    geo = rec.get("collection_event", {}).get("country", "")
    if seq:
        fasta_lines.append(f">{rec_id}|{sp}|{geo}")
        fasta_lines.append(seq)

with open("bold_gulf_of_guinea.fasta", "w") as f:
    f.write("\n".join(fasta_lines))

print(f"\nTotal records: {len(all_records)}")
print(f"Saved: bold_gulf_of_guinea.fasta")
Script 3 — Merge and build your local reference library
After Sanger sequencing your own samples, use this to add them and create a formatted FASTA + taxonomy file ready for QIIME2 or DADA2.
# build_library.py
# Run: python build_library.py

from Bio import SeqIO
from Bio.SeqRecord import SeqRecord
from Bio.Seq import Seq
import os, csv

# ---------------------------------------------------
# ADD YOUR NEW SANGER SEQUENCES HERE
# Copy the sequence text from your .ab1 / .seq file
# ---------------------------------------------------
YOUR_SEQUENCES = [
    {
        "id":        "NGA_2025_001",
        "species":   "Ethmalosa fimbriata",
        "marker":    "12S",
        "location":  "Lagos Lagoon",
        "lat":       6.453,
        "lon":       3.396,
        "date":      "2025-03-15",
        "voucher":   "NIOMR-LOS-001",
        "collector": "Your Name",
        "sequence":  "ACGTACGT..."   # paste your sequence here
    },
    # add more entries below...
]

# ---------------------------------------------------
# MERGE WITH EXISTING BOLD FASTA (from Script 2)
# ---------------------------------------------------
os.makedirs("nigeria_reflib/12S", exist_ok=True)

new_recs = []
metadata = []

for s in YOUR_SEQUENCES:
    if not s["sequence"] or s["sequence"] == "ACGTACGT...":
        continue
    rec = SeqRecord(
        Seq(s["sequence"]),
        id=s["id"],
        description=(f"{s['species']}|{s['marker']}|"
                     f"{s['location']}|{s['lat']},{s['lon']}|"
                     f"{s['date']}|{s['voucher']}")
    )
    new_recs.append(rec)
    metadata.append({**s, "length": len(s["sequence"])})

# Append to existing library
fasta_out = "nigeria_reflib/12S/nigeria_12S.fasta"
SeqIO.write(new_recs, fasta_out, "fasta")

# Write taxonomy file (QIIME2 compatible)
with open("nigeria_reflib/12S/taxonomy.tsv", "w", newline="") as f:
    w = csv.writer(f, delimiter="\t")
    w.writerow(["Feature ID", "Taxon", "Marker", "Country"])
    for s in metadata:
        w.writerow([s["id"], s["species"], s["marker"], "Nigeria"])

# Voucher registry
with open("nigeria_reflib/voucher_registry.csv", "w", newline="") as f:
    if metadata:
        w = csv.DictWriter(f, fieldnames=metadata[0].keys())
        w.writeheader(); w.writerows(metadata)

print(f"Added {len(new_recs)} sequences to library")
print(f"Files: {fasta_out}")
print(f"       nigeria_reflib/12S/taxonomy.tsv")
print(f"       nigeria_reflib/voucher_registry.csv")
Script 4 — BLAST your eDNA sequences against local library
Once you have reference sequences, use this to identify unknowns from your eDNA samples locally — no internet needed.
# blast_local.py
# Requires BLAST+ installed: https://ftp.ncbi.nlm.nih.gov/blast/executables/blast+/LATEST/
# Run: python blast_local.py

import subprocess, os

REF_FASTA  = "nigeria_reflib/12S/nigeria_12S.fasta"
QUERY_FILE = "my_edna_sequences.fasta"   # your unknown eDNA sequences
DB_NAME    = "nigeria_reflib/12S/blast_db"
OUTPUT     = "blast_results.tsv"

# Step 1: Build BLAST database from reference library
subprocess.run([
    "makeblastdb", "-in", REF_FASTA,
    "-dbtype", "nucl", "-out", DB_NAME, "-parse_seqids"
], check=True)

# Step 2: Run BLAST
subprocess.run([
    "blastn", "-query", QUERY_FILE,
    "-db", DB_NAME, "-out", OUTPUT,
    "-outfmt", "6 qseqid sseqid pident length evalue bitscore stitle",
    "-perc_identity", "95",
    "-evalue", "1e-10",
    "-num_threads", "4"
], check=True)

print(f"Results saved: {OUTPUT}")
print("Columns: query | match | %identity | length | e-value | bitscore | description")

A practical 5-step path from your laptop to a published reference library — no lab infrastructure needed to start.

1
Install Python tools (30 minutes, today)
Install Python 3.10+, then run: pip install biopython pandas requests openpyxl. These four libraries are all you need for Scripts 1–3.
Download Python BioPython docs
2
Mine existing databases (1–2 days, no fieldwork)
Run Script 1 (NCBI) and Script 2 (BOLD). This gives you the exact picture of what already exists. You will likely find that 30–50% of common commercial species already have some sequences globally, but very few from Nigeria specifically. That gap document is the foundation of your grant proposal.
BOLD public portal NCBI Nucleotide FishBase species list
3
Fish market voucher collection (first field phase)
Lagos fish markets (Mile 12, Oyingbo, Badore) land 40–60 species daily. Equipment needed: cooler, scissors, 2ml microtubes, 95% ethanol, a ruler, a smartphone camera. Photograph each specimen fully before taking a 0.5cm fin clip. Label tubes with date, location, and a sequential ID. No boats, no permits for specimens purchased at market. A single morning visit can yield 20–30 new tissue samples covering some of your highest-priority gap species.
4
Sanger sequencing and BOLD deposit
Send tissue samples to a sequencing facility. Macrogen Korea is the most cost-effective (~$4–6 per sample) with turnaround of 3–5 days. When sequences return, run Script 3 to organize them. Then create a free BOLD project and upload — your sequences immediately become publicly searchable by any lab in the world running metabarcoding on West African samples.
Macrogen sequencing Create BOLD project NCBI submission portal
5
Publish the gap analysis as a data paper
A data paper describing the library (even 30–50 species) is publishable in journals like Scientific Data (Nature), Biodiversity Data Journal, or ZooKeys. The combination of: (1) a systematic gap analysis of Nigerian coastal species in global databases, (2) new voucher-backed sequences filling priority gaps, and (3) a publicly available BOLD project constitutes a complete, citable contribution. This also serves as the evidence base for larger TETFUND or GCRF grant applications.
